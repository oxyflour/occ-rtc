cmake_minimum_required(VERSION 3.24)
project(occ_rtc VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This module currently targets Windows x64 / MSVC.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
add_compile_options(/MP)
add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/cmake/mbedtls_user_config.h")
set(OCCT_ROOT "C:/Programs/OpenCASCADE-7.7.0-vc14-64/opencascade-7.7.0" CACHE PATH "OCCT 7.7 installation")
set(TCL_ROOT "${OCCT_ROOT}/../tcltk-86-64" CACHE PATH "Tcl supplied with OCCT")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/mbedtls EXCLUDE_FROM_ALL)
add_library(MbedTLS::MbedTLS INTERFACE IMPORTED GLOBAL)
set_target_properties(MbedTLS::MbedTLS PROPERTIES INTERFACE_LINK_LIBRARIES "mbedtls;mbedx509;mbedcrypto"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mbedtls/include")
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
# libSRTP's built-in AES128/SHA1 backend is sufficient for the negotiated SRTP suite.
set(ENABLE_MBEDTLS OFF CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_WITH_WARNINGS OFF CACHE BOOL "" FORCE)
set(ENABLE_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/libdatachannel EXCLUDE_FROM_ALL)
add_library(TKDrawRTC SHARED src/Plugin.cxx src/Session.cxx src/Encoder.cxx)
target_compile_definitions(TKDrawRTC PRIVATE WIN32_LEAN_AND_MEAN= NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(TKDrawRTC PRIVATE /W4 /EHsc /utf-8 /bigobj)
target_include_directories(TKDrawRTC PRIVATE "${OCCT_ROOT}/inc" "${TCL_ROOT}/include"
  third_party/libdatachannel/deps/json/single_include third_party/httplib)
target_link_directories(TKDrawRTC PRIVATE "${OCCT_ROOT}/win64/vc14/lib" "${TCL_ROOT}/lib")
target_link_libraries(TKDrawRTC PRIVATE datachannel-static TKDraw TKViewerTest TKV3d TKOpenGl
  TKService TKBRep TKMath TKernel tcl86 mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 ws2_32)
file(READ "${OCCT_ROOT}/src/DrawResources/DrawPlugin" DRAW_PLUGINS)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/resources")
file(WRITE "${CMAKE_BINARY_DIR}/resources/DrawPlugin" "${DRAW_PLUGINS}\nDRAWRTC : TKDrawRTC\n")
add_custom_target(drawrtc_web
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/web")
add_dependencies(TKDrawRTC drawrtc_web)
